        // Sound Synthesis using Tone.js
        function initAudio() {
            try {
                synth = new Tone.Synth().toDestination();

                console.log("Audio Init Error", e);
            }
        }

        function playSound(type) {
            if (isAudioMuted || !synth) return;
            Tone.start();
            
            const now = Tone.now();
            if (type === 'correct') {
                synth.triggerAttackRelease("C5", "8n", now);
                synth.triggerAttackRelease("E5", "8n", now + 0.15);
                synth.triggerAttackRelease("G5", "8n", now + 0.3);
            } else if (type === 'wrong') {
                synth.triggerAttackRelease("A3", "8n", now);
                synth.triggerAttackRelease("F3", "8n", now + 0.2);
            } else if (type === 'victory') {
                synth.triggerAttackRelease("C4", "8n", now);
                synth.triggerAttackRelease("E4", "8n", now + 0.2);
                synth.triggerAttackRelease("G4", "8n", now + 0.4);
                synth.triggerAttackRelease("C5", "4n", now + 0.6);
            } else if (type === 'click') {
                synth.triggerAttackRelease("G4", "16n", now);
            }
        }

        function toggleAudio() {
            isAudioMuted = !isAudioMuted;
            const icon = document.getElementById('soundIcon');
            if (isAudioMuted) {
                icon.className = 'fa-solid fa-volume-xmark text-red-400';
            } else {
                icon.className = 'fa-solid fa-volume-high text-amber-400';
                playSound('click');
            }
        }

                    <div class="my-3">
                        <i class="fa-solid ${island.icon} text-3xl md:text-4xl mb-2"></i>
                        <h4 class="font-bold text-sm md:text-base text-white">${island.name}</h4>
                        <p class="text-[11px] opacity-80 mt-1">${island.skill}</p>
                    </div>

                    <!-- Interactive Chest Action Button -->
                    <button onclick="openIslandQuestion(${idx})" ${isLocked ? 'disabled' : ''} 
                        class="w-full mt-3 py-2 px-3 rounded-xl font-bold text-xs flex items-center justify-center gap-1.5 transition ${
                            isCurrent ? 'bg-gradient-to-r from-amber-500 to-yellow-500 text-slate-950 hover:brightness-110 shadow-lg' : 
                            isPassed ? 'bg-emerald-600/30 text-emerald-300 hover:bg-emerald-600/50' : 'bg-slate-800 text-slate-600 cursor-not-allowed'
                        }">
                        <i class="fa-solid ${isPassed ? 'fa-box-open' : 'fa-box'}"></i>
                        <span>${isPassed ? 'إعادة الإجابة' : isCurrent ? 'افتح الصندوق' : 'مغلقة'}</span>
                    </button>
                `;

                container.appendChild(islandCard);
            });

            // Update stats top bar
            document.getElementById('currentIslandDisplay').innerText = Math.min(currentIslandIndex + 1, 10);
            document.getElementById('scoreDisplay').innerText = score;
        }

        // Start Game Handler
        function startGame() {
            const nameInput = document.getElementById('playerNameInput').value.trim();
            if (nameInput) {
                playerName = nameInput;
            }
            initAudio();
            playSound('click');
            document.getElementById('startModal').classList.add('hidden');
            renderMap();
        }

        // Open Question Modal for selected Island
        function openIslandQuestion(index) {
            if (index > currentIslandIndex) return; // Prevent clicking locked islands

            playSound('click');
            const data = islandsData[index];

            document.getElementById('modalIslandBadge').innerText = `الجزيرة ${data.id}`;
            document.getElementById('modalSkillTitle').innerText = `مهارة: ${data.skill}`;
            document.getElementById('questionText').innerText = data.question;

            const optionsContainer = document.getElementById('optionsContainer');
            optionsContainer.innerHTML = '';

            data.options.forEach((opt, optIdx) => {
                const btn = document.createElement('button');
                btn.className = "p-4 bg-slate-700/70 hover:bg-amber-500 hover:text-slate-950 text-white font-bold rounded-xl border border-slate-600 transition text-right flex items-center justify-between group";
                btn.onclick = () => checkAnswer(index, optIdx);
                btn.innerHTML = `
                    <span>${opt}</span>
                    <i class="fa-solid fa-circle-chevron-left opacity-0 group-hover:opacity-100 transition"></i>
                `;
                optionsContainer.appendChild(btn);
            });

            // Reset feedback banner
            const feedback = document.getElementById('feedbackBanner');
            feedback.className = "mt-4 p-3 rounded-xl text-center font-bold text-sm hidden";

            document.getElementById('questionModal').classList.remove('hidden');
        }

        // Check selected answer
        function checkAnswer(islandIdx, selectedIdx) {
            const data = islandsData[islandIdx];
            const feedback = document.getElementById('feedbackBanner');
            feedback.classList.remove('hidden');

            if (selectedIdx === data.correct) {
                // Correct Answer
                playSound('correct');
                feedback.className = "mt-4 p-3 rounded-xl text-center font-bold text-sm bg-emerald-500/20 text-emerald-300 border border-emerald-500/50";
                feedback.innerHTML = `<i class="fa-solid fa-circle-check ml-1"></i> إجابة صحيحة وممتازة! أحسنت يا بطل!`;

                // Add score if answered first time on current island
                if (islandIdx === currentIslandIndex) {
                    score += 100;
                    currentIslandIndex++;
                }

                setTimeout(() => {
                    document.getElementById('questionModal').classList.add('hidden');
                    renderMap();

                    // Check Victory
                    if (currentIslandIndex >= islandsData.length) {
                        showVictoryCertificate();
                    }
                }, 1400);

            } else {
                // Wrong Answer
                playSound('wrong');
                feedback.className = "mt-4 p-3 rounded-xl text-center font-bold text-sm bg-rose-500/20 text-rose-300 border border-rose-500/50 animate-shake";
                feedback.innerHTML = `<i class="fa-solid fa-circle-xmark ml-1"></i> إجابة خاطئة! حاول مرة أخرى وحافظ على تركيزك.`;
            }
        }

        // Victory Screen & Certificate
        function showVictoryCertificate() {
            playSound('victory');
            
            // Launch Confetti Effect
            if (typeof confetti === 'function') {
                confetti({
                    particleCount: 150,
                    spread: 80,
                    origin: { y: 0.6 }
                });
            }

            document.getElementById('certStudentName').innerText = playerName;
            document.getElementById('certScore').innerText = `${score} / 1000`;
            document.getElementById('certificateModal').classList.remove('hidden');
        }

        // Print Certificate
        function printCertificate() {
            window.print();
        }

        // Restart Game
        function restartGame() {
            currentIslandIndex = 0;
            score = 0;
            document.getElementById('certificateModal').classList.add('hidden');
            renderMap();
        }

        // Initialize Map on window load
        window.onload = function() {
            renderMap();
        };
    </script>
